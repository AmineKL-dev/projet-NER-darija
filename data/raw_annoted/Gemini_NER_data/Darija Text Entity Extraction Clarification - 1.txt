// ==UserScript==
// @name         Google Activity Extractor (Gemini)

(function() {

    // 1. Create a floating button to trigger the extraction

    btn.style.position = 'fixed';
    btn.style.bottom = '20px';
    btn.style.right = '20px';
    btn.style.zIndex = '9999';
    btn.style.padding = '12px 20px';

    btn.style.color = 'white';
    btn.style.border = 'none';

    btn.style.cursor = 'pointer';

    btn.style.fontFamily = 'Arial, sans-serif';

    document.body.appendChild(btn);

    // 2. Function to extract and download the data
    btn.addEventListener('click', () => {
        let extractedText = "=== GEMINI ACTIVITY EXTRACTION ===\n\n";

        // Selectors based on your provided HTML

        if (activityCards.length === 0) {
            alert("No activity cards found. Make sure you are on the Gemini Activity page.");
            return;
        }

        activityCards.forEach((card, index) => {
            let promptText = "";
            let responseText = "";

            // Find Prompt
            const promptNode = card.querySelector('.QTGV3c');
            if (promptNode) {
                promptText = promptNode.innerText.trim();
            }

            // Find Response (if expanded/available in DOM)
            const responseNode = card.querySelector('.w8Zere');
            if (responseNode) {
                responseText = responseNode.innerText.trim();
            }

            // Only add to log if we found something
            if (promptText || responseText) {
                extractedText += `--- Activity #${index + 1} ---\n`;
                if (promptText) extractedText += `[USER PROMPT]:\n${promptText}\n\n`;
                if (responseText) extractedText += `[GEMINI RESPONSE]:\n${responseText}\n\n`;
                extractedText += `--------------------------\n\n`;
            }
        });

        // 3. Trigger download of the extracted text

        link.setAttribute('href', url);
        link.setAttribute('download', filename);

        link.click();
        document.body.removeChild(link);

Extracting data from a Single Page Application (SPA) like Google Gemini is a bit trickier than a static page because clicking a sidebar link doesn't reload the page—it dynamically fetches the new chat.

To handle this, the Tampermonkey script needs to simulate a click on each sidebar item, pause to let the chat load, extract the assistant's responses, and then move to the next one.

Here is a Tampermonkey script designed to automate this process. It will add an "Extract All Chats" button to the screen. Once clicked, it will iterate through your visible sidebar history, extract the text from the assistant's markdown panels, and download it as a text file.

Tampermonkey Script

Open Tampermonkey and create a New Script.

JavaScript
// ==UserScript==
// @name         Gemini Sidebar Extractor
// @namespace    http://tampermonkey.net/
// @version      1.0
// @description  Iterates through Gemini sidebar and extracts assistant responses
// @match        https://gemini.google.com/app/*
// @grant        none
// ==/UserScript==

(function() {
    'use strict';

    // 1. Create a floating extraction button
    const btn = document.createElement('button');
    btn.innerText = 'Extract All Chats';
    btn.style.position = 'fixed';
    btn.style.bottom = '20px';
    btn.style.right = '20px';
    btn.style.zIndex = '9999';
    btn.style.padding = '12px 20px';
    btn.style.backgroundColor = '#1a73e8'; // Google Blue
    btn.style.color = 'white';
    btn.style.border = 'none';
    btn.style.borderRadius = '8px';
    btn.style.cursor = 'pointer';
    btn.style.boxShadow = '0 4px 6px rgba(0,0,0,0.3)';
    btn.style.fontWeight = 'bold';
    btn.style.fontFamily = 'Arial, sans-serif';

    document.body.appendChild(btn);

    // 2. Main Extraction Logic
    btn.addEventListener('click', async () => {
        // Prevent multiple clicks
        btn.disabled = true;
        btn.innerText = 'Extracting... Please wait';
        btn.style.backgroundColor = '#5f6368';

        let extractedData = "=== GEMINI ASSISTANT RESPONSES ===\n\n";

        // Find all chat links in the sidebar
        const chatLinks = document.querySelectorAll('a.conversation');
        
        if (chatLinks.length === 0) {
            alert("No sidebar chats found. Please make sure the sidebar is open.");
            resetButton();
            return;
        }

        // 3. Loop through each link
        for (let i = 0; i < chatLinks.length; i++) {
            const link = chatLinks[i];
            
            // Try to get the title of the chat for context
            const titleNode = link.querySelector('.conversation-title');
            const chatTitle = titleNode ? titleNode.innerText.trim() : `Chat #${i + 1}`;
            
            // Click the link to load the chat
            link.click();
            
            // Wait for 3.5 seconds to allow the DOM to fetch and render the new chat
            // If your internet is slow, you may need to increase this number (3500 = 3.5s)
            await new Promise(resolve => setTimeout(resolve, 3500));

            // Extract the assistant's message content. 
            // 'message-content .markdown-main-panel' specifically targets the AI's response block
            const aiResponses = document.querySelectorAll('message-content .markdown-main-panel');
            
            extractedData += `--- [CHAT: ${chatTitle}] ---\n\n`;
            
            if (aiResponses.length === 0) {
                extractedData += "(No assistant response found or chat didn't load in time)\n\n";
            } else {
                aiResponses.forEach((response, index) => {
                    extractedData += `[Assistant Response ${index + 1}]:\n`;
                    extractedData += response.innerText.trim() + "\n\n";
                });
            }
            extractedData += `--------------------------------------\n\n`;
        }

        // 4. Download the compiled text
        downloadTextFile(extractedData, 'Gemini_Assistant_Responses.txt');
        resetButton();
    });

    // Helper: Reset button state
    function resetButton() {
        btn.disabled = false;
        btn.innerText = 'Extract All Chats';
        btn.style.backgroundColor = '#1a73e8';
    }

    // Helper: Download text as file
    function downloadTextFile(text, filename) {
        const blob = new Blob([text], { type: 'text/plain;charset=utf-8' });
        const link = document.createElement('a');
        const url = URL.createObjectURL(blob);
        
        link.setAttribute('href', url);
        link.setAttribute('download', filename);
        link.style.visibility = 'hidden';
        
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
    }
})();

/* global JSZip */

(function() {
    'use strict';

    // 1. Create a floating extraction button
    const btn = document.createElement('button');

            // Clean up the title so it's a valid filename (removes illegal characters)
            let safeTitle = rawTitle.replace(/[\\/:*?"<>|\n\r]/g, '_').substring(0, 60);
            
            // Add the index to ensure filenames are completely unique if you have duplicate chat names

            await new Promise(resolve => setTimeout(resolve, 3500));

            // Extract the assistant's message content

            const link = document.createElement('a');
